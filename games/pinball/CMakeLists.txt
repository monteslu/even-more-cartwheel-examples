# pinball (working title): original pinball tables with a ball solver of its own, lit and rendered in 3D.
#   pinball            the game
#   pinball_auto       the bot plays it (for hosts that send no input)
#   pinball_bench      the perf gate's scene: three balls on the busiest table, all the time (carts read no environment)
#   pinball_physcheck  the physics question: the game's solver against the engine's Box2D and Box3D (native only)
#   pinball_soak       the rules and the bot with no window and no GPU (native only)
#   pinball_tablemap   a top-down picture of a table's physics and the bot's ball (native only, for laying out tables)
set(PB_SIM src/phys.c src/tables.c src/rules.c src/bot.c src/lights.c)
set(PB_SRC src/main.c src/view.c src/backbox.c src/geo.c src/art.c src/dmd.c src/sound.c ${PB_SIM})
set(CW_PROGRAMS ${CMAKE_CURRENT_SOURCE_DIR}/programs/pinball.txt)
cw_add_app(pinball SOURCES ${PB_SRC} PACK ${CMAKE_CURRENT_SOURCE_DIR}/assets/roots.txt)
cw_add_app(pinball_auto SOURCES ${PB_SRC} PACK ${CMAKE_CURRENT_SOURCE_DIR}/assets/roots.txt)
target_compile_definitions(pinball_auto PRIVATE PINBALL_AUTOPILOT=1)
add_dependencies(pinball_auto_pack pinball_pack)
cw_add_app(pinball_bench SOURCES ${PB_SRC} PACK ${CMAKE_CURRENT_SOURCE_DIR}/assets/roots.txt)
target_compile_definitions(pinball_bench PRIVATE PINBALL_AUTOPILOT=1 PINBALL_BENCH=1 PINBALL_BENCH_TABLE=1)
add_dependencies(pinball_bench_pack pinball_pack)
if(NOT CW_CART)
  add_executable(pinball_physcheck tools/physcheck.c src/phys.c)
  target_link_libraries(pinball_physcheck PRIVATE cartwheel)
  add_executable(pinball_soak src/soak.c ${PB_SIM})
  target_link_libraries(pinball_soak PRIVATE cartwheel)
  add_executable(pinball_tablemap tools/tablemap.c ${PB_SIM})
  target_link_libraries(pinball_tablemap PRIVATE cartwheel)
  target_include_directories(pinball_tablemap PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../deps/cartwheel-sdk/linux-x64/deps/stb)
endif()
if(NOT CW_CART)
  add_executable(pinball_logo tools/logo.c src/dmd.c)
  target_link_libraries(pinball_logo PRIVATE m)
  target_include_directories(pinball_logo PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../deps/cartwheel-sdk/linux-x64/deps/stb)
endif()
if(NOT CW_CART)
  # tests/pinball-drain.mjs: when the flipper buttons work; the _oldrule build is its control (the rule before)
  add_executable(pinball_flipcheck tools/flipcheck.c ${PB_SIM})
  target_link_libraries(pinball_flipcheck PRIVATE cartwheel)
  add_executable(pinball_flipcheck_oldrule tools/flipcheck.c ${PB_SIM})
  target_link_libraries(pinball_flipcheck_oldrule PRIVATE cartwheel)
  target_compile_definitions(pinball_flipcheck_oldrule PRIVATE PINBALL_OLD_FLIPPER_RULE=1)
endif()
if(NOT CW_CART)
  # tests/pinball-flash.mjs: every light's level over whole games against the flash rule; the _oldstrobe builds are its
  # control (the lights as they were: blinking inserts, flickering flashers), which must fail
  add_executable(pinball_flashcheck tools/flashcheck.c ${PB_SIM})
  target_link_libraries(pinball_flashcheck PRIVATE cartwheel)
  add_executable(pinball_flashcheck_oldstrobe tools/flashcheck.c ${PB_SIM})
  target_link_libraries(pinball_flashcheck_oldstrobe PRIVATE cartwheel)
  target_compile_definitions(pinball_flashcheck_oldstrobe PRIVATE PINBALL_OLD_STROBE=1)
  cw_add_app(pinball_bench_oldstrobe SOURCES ${PB_SRC} PACK ${CMAKE_CURRENT_SOURCE_DIR}/assets/roots.txt)
  target_compile_definitions(pinball_bench_oldstrobe PRIVATE PINBALL_AUTOPILOT=1 PINBALL_BENCH=1 PINBALL_BENCH_TABLE=1 PINBALL_OLD_STROBE=1)
  add_dependencies(pinball_bench_oldstrobe_pack pinball_pack)
  # tests/pinball-rules.mjs: each table's modes, toy and wizard mode, driven through the rules' own switch events; the
  # _oneset build (every table on one shared rule set, no toys: the rules as they were) is its control
  add_executable(pinball_rulecheck tools/rulecheck.c ${PB_SIM})
  target_link_libraries(pinball_rulecheck PRIVATE cartwheel)
  add_executable(pinball_rulecheck_oneset tools/rulecheck.c ${PB_SIM})
  target_link_libraries(pinball_rulecheck_oneset PRIVATE cartwheel)
  target_compile_definitions(pinball_rulecheck_oneset PRIVATE PINBALL_RULES_ONE_SET=1)
endif()
