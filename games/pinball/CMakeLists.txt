# pinball (working title): original pinball tables with a ball solver of its own, lit and rendered in 3D.
#   pinball            the game
#   pinball_auto       the bot plays it (for hosts that send no input)
#   pinball_bench      the perf gate's scene: three balls on the busiest table, all the time (carts read no environment)
#   pinball_physcheck  the physics question: the game's solver against the engine's Box2D and Box3D (native only)
#   pinball_soak       the rules and the bot with no window and no GPU (native only)
#   pinball_tablemap   a top-down picture of a table's physics and the bot's ball (native only, for laying out tables)
set(PB_SIM src/phys.c src/tables.c src/rules.c src/bot.c)
set(PB_SRC src/main.c src/view.c src/geo.c src/art.c src/dmd.c src/sound.c ${PB_SIM})
set(CW_PROGRAMS ${CMAKE_CURRENT_SOURCE_DIR}/programs/pinball.txt)
cw_add_app(pinball SOURCES ${PB_SRC} PACK ${CMAKE_CURRENT_SOURCE_DIR}/assets/roots.txt)
cw_add_app(pinball_auto SOURCES ${PB_SRC} PACK ${CMAKE_CURRENT_SOURCE_DIR}/assets/roots.txt)
target_compile_definitions(pinball_auto PRIVATE PINBALL_AUTOPILOT=1)
add_dependencies(pinball_auto_pack pinball_pack)
cw_add_app(pinball_bench SOURCES ${PB_SRC} PACK ${CMAKE_CURRENT_SOURCE_DIR}/assets/roots.txt)
target_compile_definitions(pinball_bench PRIVATE PINBALL_AUTOPILOT=1 PINBALL_BENCH=1 PINBALL_BENCH_TABLE=1)
add_dependencies(pinball_bench_pack pinball_pack)
if(NOT CW_CART)
  add_executable(pinball_physcheck tools/physcheck.c src/phys.c)
  target_link_libraries(pinball_physcheck PRIVATE cartwheel)
  add_executable(pinball_soak src/soak.c ${PB_SIM})
  target_link_libraries(pinball_soak PRIVATE cartwheel)
  add_executable(pinball_tablemap tools/tablemap.c ${PB_SIM})
  target_link_libraries(pinball_tablemap PRIVATE cartwheel)
  target_include_directories(pinball_tablemap PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../deps/cartwheel-sdk/linux-x64/deps/stb)
endif()
if(NOT CW_CART)
  add_executable(pinball_logo tools/logo.c src/dmd.c)
  target_link_libraries(pinball_logo PRIVATE m)
  target_include_directories(pinball_logo PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../deps/cartwheel-sdk/linux-x64/deps/stb)
endif()
