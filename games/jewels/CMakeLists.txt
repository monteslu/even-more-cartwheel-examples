# jewels (working title): a calm, untimed match-three of six cut stones, ray traced through their facets.
#   jewels             the game
#   jewels_auto        the bot plays it through the controls (pad, touch, mouse and drags in turn)
#   jewels_bench       the perf gate's scene: the bot picks the biggest clear every move (carts read no environment)
#   jewels_soak        the rules, the game and the bot's moves with no window and no GPU (native only)
#   jewels_rules_check the C game against traces of the original Lua game (tests/jewels-rules.mjs; native only)
set(JW_SIM src/rules.c src/game.c)
set(JW_SRC src/main.c src/render.c src/shaders.c src/cuts.c src/fx.c src/sound.c ${JW_SIM})
set(CW_PROGRAMS ${CMAKE_CURRENT_SOURCE_DIR}/programs/jewels.txt)
cw_add_app(jewels SOURCES ${JW_SRC} PACK ${CMAKE_CURRENT_SOURCE_DIR}/assets/roots.txt)
cw_add_app(jewels_auto SOURCES ${JW_SRC} PACK ${CMAKE_CURRENT_SOURCE_DIR}/assets/roots.txt)
target_compile_definitions(jewels_auto PRIVATE JEWELS_AUTOPILOT=1)
add_dependencies(jewels_auto_pack jewels_pack)
cw_add_app(jewels_bench SOURCES ${JW_SRC} PACK ${CMAKE_CURRENT_SOURCE_DIR}/assets/roots.txt)
target_compile_definitions(jewels_bench PRIVATE JEWELS_AUTOPILOT=1 JEWELS_BENCH=1)
add_dependencies(jewels_bench_pack jewels_pack)
if(NOT CW_CART)
  add_executable(jewels_soak src/soak.c src/fx.c src/cuts.c ${JW_SIM})
  target_link_libraries(jewels_soak PRIVATE m)
  add_executable(jewels_rules_check tools/rules_check.c ${JW_SIM})
  target_link_libraries(jewels_rules_check PRIVATE m)
endif()
