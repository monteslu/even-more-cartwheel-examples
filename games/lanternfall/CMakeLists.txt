# Lanternfall: a 2D exploration platformer (DESIGN.md).
#   lanternfall          the game
#   lanternfall_auto     the bot plays it along the checker's route (for hosts that send no input)
#   lanternfall_bench    the perf gate's scene: a warden fight in its arena, both players, every ability (carts read no
#                     environment)
#   lanternfall_worldgen builds assets/world.bin (native only)
#   lanternfall_check    proves the world: completable in ability order, no soft locks, every pickup, the gates, nothing
#                     floating; writes assets/route.bin (native only)
#   lanternfall_feel     measures the movement: jump heights, distances, coyote time, jump buffer (native only)
#   lanternfall_rules    measures the combat rules: the hold after a blow, knockback, the puff, trials, phases,
#                     charms, the flame left where you fall, each with a control (native only)
#   lanternfall_soak     the rules and the bot with no window and no GPU (native only)
set(EX_SIM src/sim.c src/player.c src/reach.c src/bot.c src/world.c src/regions.c)
set(EX_SRC src/main.c src/view.c src/art.c src/hud.c src/audio.c ${EX_SIM})
cw_add_app(lanternfall SOURCES ${EX_SRC} PACK ${CMAKE_CURRENT_SOURCE_DIR}/assets/roots.txt)
cw_add_app(lanternfall_auto SOURCES ${EX_SRC} PACK ${CMAKE_CURRENT_SOURCE_DIR}/assets/roots.txt)
target_compile_definitions(lanternfall_auto PRIVATE LANTERNFALL_AUTOPILOT=1)
add_dependencies(lanternfall_auto_pack lanternfall_pack)
cw_add_app(lanternfall_bench SOURCES ${EX_SRC} PACK ${CMAKE_CURRENT_SOURCE_DIR}/assets/roots.txt)
target_compile_definitions(lanternfall_bench PRIVATE LANTERNFALL_AUTOPILOT=1 LANTERNFALL_BENCH=2 LANTERNFALL_AUTO_PLAYERS=2)
add_dependencies(lanternfall_bench_pack lanternfall_pack)
foreach(t lanternfall lanternfall_auto lanternfall_bench)
  target_include_directories(${t} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
endforeach()
if(NOT CW_CART)
  find_package(Threads REQUIRED)
  add_executable(lanternfall_worldgen tools/worldgen.c src/worldgen.c src/world.c src/regions.c)
  target_compile_definitions(lanternfall_worldgen PRIVATE LANTERNFALL_NO_ENGINE=1)
  target_link_libraries(lanternfall_worldgen PRIVATE m)
  add_executable(lanternfall_check tools/check.c src/reach.c src/player.c src/world.c src/regions.c)
  target_compile_definitions(lanternfall_check PRIVATE LANTERNFALL_NO_ENGINE=1)
  target_link_libraries(lanternfall_check PRIVATE m Threads::Threads)
  add_executable(lanternfall_feel tools/feel.c src/player.c src/world.c src/regions.c)
  target_compile_definitions(lanternfall_feel PRIVATE LANTERNFALL_NO_ENGINE=1)
  target_link_libraries(lanternfall_feel PRIVATE m)
  add_executable(lanternfall_rules tools/rules.c src/sim.c src/player.c src/world.c src/regions.c)
  target_compile_definitions(lanternfall_rules PRIVATE LANTERNFALL_NO_ENGINE=1)
  target_link_libraries(lanternfall_rules PRIVATE cartwheel)
  add_executable(lanternfall_soak src/soak.c ${EX_SIM})
  target_compile_definitions(lanternfall_soak PRIVATE LANTERNFALL_NO_ENGINE=1)
  target_link_libraries(lanternfall_soak PRIVATE cartwheel)
endif()
