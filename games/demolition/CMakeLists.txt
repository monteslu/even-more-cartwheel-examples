# demolition (working title): knock down structures of wood, stone, glass and steel with launched shot and charges.
#   demolition             the game
#   demolition_auto        the bot plays it through the controls (for hosts that send no input)
#   demolition_bench       the perf gate's scene: the bot on Great Keep, the collapse in view (carts read no environment)
#   demolition_bench_nobudget  the game as before the hitch fix: no fracture queue, no blast shells, no effects primed (the hitch gate's control)
#   demolition_physcheck   what the engine's Box3D can do here: rest, budget, determinism, tunnelling (native only)
#   demolition_levelcheck  every level stands, nothing floats, the bot clears it within par (native only)
#   demolition_soak        level after level with the bot, no window and no GPU (native only)
set(DM_SIM src/sim.c src/levels.c src/bot.c)
set(DM_SRC src/main.c src/view.c src/fx.c src/sound.c ${DM_SIM})
set(CW_PROGRAMS ${CMAKE_CURRENT_SOURCE_DIR}/programs/demolition.txt)
cw_add_app(demolition SOURCES ${DM_SRC} PACK ${CMAKE_CURRENT_SOURCE_DIR}/assets/roots.txt)
cw_add_app(demolition_auto SOURCES ${DM_SRC} PACK ${CMAKE_CURRENT_SOURCE_DIR}/assets/roots.txt)
target_compile_definitions(demolition_auto PRIVATE DEMOLITION_AUTOPILOT=1)
add_dependencies(demolition_auto_pack demolition_pack)
cw_add_app(demolition_bench SOURCES ${DM_SRC} PACK ${CMAKE_CURRENT_SOURCE_DIR}/assets/roots.txt)
target_compile_definitions(demolition_bench PRIVATE DEMOLITION_AUTOPILOT=1 DEMOLITION_BENCH=1)
add_dependencies(demolition_bench_pack demolition_pack)
cw_add_app(demolition_bench_nobudget SOURCES ${DM_SRC} PACK ${CMAKE_CURRENT_SOURCE_DIR}/assets/roots.txt)
target_compile_definitions(demolition_bench_nobudget PRIVATE DEMOLITION_AUTOPILOT=1 DEMOLITION_BENCH=1 DEMOLITION_NO_BUDGET=1)
add_dependencies(demolition_bench_nobudget_pack demolition_pack)
if(NOT CW_CART)
  add_executable(demolition_physcheck tools/physcheck.c)
  target_link_libraries(demolition_physcheck PRIVATE cartwheel)
  add_executable(demolition_levelcheck tools/levelcheck.c ${DM_SIM})
  target_include_directories(demolition_levelcheck PRIVATE src)
  target_link_libraries(demolition_levelcheck PRIVATE cartwheel)
  add_executable(demolition_soak src/soak.c ${DM_SIM})
  target_link_libraries(demolition_soak PRIVATE cartwheel)
endif()
