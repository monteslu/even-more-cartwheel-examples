# explorer (working title): a 2D exploration platformer (DESIGN.md).
#   explorer          the game
#   explorer_auto     the bot plays it along the checker's route (for hosts that send no input)
#   explorer_bench    the perf gate's scene: a warden fight in its arena, both players, every ability (carts read no
#                     environment)
#   explorer_worldgen builds assets/world.bin (native only)
#   explorer_check    proves the world: completable in ability order, no soft locks, every pickup, the gates, nothing
#                     floating; writes assets/route.bin (native only)
#   explorer_feel     measures the movement: jump heights, distances, coyote time, jump buffer (native only)
#   explorer_soak     the rules and the bot with no window and no GPU (native only)
set(EX_SIM src/sim.c src/player.c src/reach.c src/bot.c src/world.c src/regions.c)
set(EX_SRC src/main.c src/view.c src/art.c src/hud.c src/audio.c ${EX_SIM})
cw_add_app(explorer SOURCES ${EX_SRC} PACK ${CMAKE_CURRENT_SOURCE_DIR}/assets/roots.txt)
cw_add_app(explorer_auto SOURCES ${EX_SRC} PACK ${CMAKE_CURRENT_SOURCE_DIR}/assets/roots.txt)
target_compile_definitions(explorer_auto PRIVATE EXPLORER_AUTOPILOT=1)
add_dependencies(explorer_auto_pack explorer_pack)
cw_add_app(explorer_bench SOURCES ${EX_SRC} PACK ${CMAKE_CURRENT_SOURCE_DIR}/assets/roots.txt)
target_compile_definitions(explorer_bench PRIVATE EXPLORER_AUTOPILOT=1 EXPLORER_BENCH=2 EXPLORER_AUTO_PLAYERS=2)
add_dependencies(explorer_bench_pack explorer_pack)
# the 2D renderer's HDR post chain, emission, soft shadows, GI, water and distortion (cartwheel-engine v0.37):
# used when the engine has them, else the game draws as before
set(_r2d "")
foreach(_c "${CMAKE_SOURCE_DIR}/include/cartwheel/render2d.h" "${cartwheel_DIR}/../../../include/cartwheel/render2d.h")
  if(NOT _r2d AND EXISTS "${_c}")
    set(_r2d "${_c}")
  endif()
endforeach()
set(EX_R2D_V2 OFF)
if(_r2d)
  file(READ "${_r2d}" _r2d_text)
  string(FIND "${_r2d_text}" "cw_r2d_post" _at)
  if(NOT _at EQUAL -1)
    set(EX_R2D_V2 ON)
  endif()
endif()
foreach(t explorer explorer_auto explorer_bench)
  if(EX_R2D_V2)
    target_compile_definitions(${t} PRIVATE EXPLORER_R2D_V2=1)
  endif()
  target_include_directories(${t} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
endforeach()
if(NOT CW_CART)
  find_package(Threads REQUIRED)
  add_executable(explorer_worldgen tools/worldgen.c src/worldgen.c src/world.c src/regions.c)
  target_compile_definitions(explorer_worldgen PRIVATE EXPLORER_NO_ENGINE=1)
  target_link_libraries(explorer_worldgen PRIVATE m)
  add_executable(explorer_check tools/check.c src/reach.c src/player.c src/world.c src/regions.c)
  target_compile_definitions(explorer_check PRIVATE EXPLORER_NO_ENGINE=1)
  target_link_libraries(explorer_check PRIVATE m Threads::Threads)
  add_executable(explorer_feel tools/feel.c src/player.c src/world.c src/regions.c)
  target_compile_definitions(explorer_feel PRIVATE EXPLORER_NO_ENGINE=1)
  target_link_libraries(explorer_feel PRIVATE m)
  add_executable(explorer_soak src/soak.c ${EX_SIM})
  target_compile_definitions(explorer_soak PRIVATE EXPLORER_NO_ENGINE=1)
  target_link_libraries(explorer_soak PRIVATE cartwheel)
endif()
