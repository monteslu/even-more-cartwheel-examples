# beatsync: a house-music rhythm game on three buttons, drawn as neon characters (README.md, docs/DESIGN.md).
#   beatsync        the game
#   beatsync_auto   the game playing itself: straight into play on Hard, every note hit from the chart (the gates,
#                   the flash sweep and the screenshots run it)
#
# The songs are baked in at build time: every songs/<id>/song.json is listed in the game (song_list.c) and its
# song.json and tracks are packed. A new song needs a rebuild (CONFIGURE_DEPENDS reruns this when one appears or goes).
# A cart with songs/index.txt among its assets lists its songs from that instead, at start (src/song.h songs_load):
# a cart folder gets songs added without a rebuild.
file(GLOB _found CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/songs/*/song.json)
list(SORT _found)
# songs/order.txt puts the songs it names first, in its order; the rest follow by name
set(_songs "")
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/songs/order.txt)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/songs/order.txt)
  file(STRINGS ${CMAKE_CURRENT_SOURCE_DIR}/songs/order.txt _order REGEX "^[^#]")
  foreach(_id ${_order})
    string(STRIP "${_id}" _id)
    set(_f ${CMAKE_CURRENT_SOURCE_DIR}/songs/${_id}/song.json)
    if(_f IN_LIST _found)
      list(APPEND _songs ${_f})
      list(REMOVE_ITEM _found ${_f})
    endif()
  endforeach()
endif()
list(APPEND _songs ${_found})
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/roots.txt _roots)
set(_ids "")
string(APPEND _roots "\n# the songs (CMakeLists.txt adds every songs/<id>/)\n")
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/songs/order.txt)
  string(APPEND _roots "songs/order.txt\n") # the game reads it at start (src/song.c songs_load)
endif()
foreach(_s ${_songs})
  get_filename_component(_dir ${_s} DIRECTORY)
  get_filename_component(_id ${_dir} NAME)
  string(APPEND _ids "  \"${_id}\",\n")
  string(APPEND _roots "songs/${_id}/song.json\n")
  file(READ ${_s} _json)
  # the track files song.json names ("tracks": { "kick": "kick.ogg", ... })
  foreach(_k kick clap bass other)
    string(REGEX MATCH "\"${_k}\" *: *\"([^\"]+)\"" _m "${_json}")
    if(NOT CMAKE_MATCH_1)
      message(FATAL_ERROR "${_s}: no tracks.${_k}")
    endif()
    string(APPEND _roots "songs/${_id}/${CMAKE_MATCH_1}\n")
  endforeach()
  # the vocals, a song may have them
  string(REGEX MATCH "\"vocals\" *: *\"([^\"]+)\"" _m "${_json}")
  if(_m)
    string(APPEND _roots "songs/${_id}/${CMAKE_MATCH_1}\n")
  endif()
endforeach()
if(NOT _songs)
  message(FATAL_ERROR "no songs/<id>/song.json: beatsync needs at least one song")
endif()
# beside roots.txt, so the packer finds every path (and the scenes' UIDs) where roots.txt does (git ignores it)
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_SOURCE_DIR}/roots.gen.txt CONTENT "${_roots}")
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/song_list.c CONTENT "/* written by CMakeLists.txt: every songs/<id>/ */\nconst char *const SONG_IDS[] = {\n${_ids}};\nconst int SONG_COUNT = (int)(sizeof SONG_IDS / sizeof SONG_IDS[0]);\n")

file(GLOB BS_SRC CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.c)
list(APPEND BS_SRC ${CMAKE_CURRENT_BINARY_DIR}/song_list.c)
set(CW_PROGRAMS ${CMAKE_CURRENT_SOURCE_DIR}/programs/beatsync.txt)
cw_add_app(beatsync SOURCES ${BS_SRC} PACK ${CMAKE_CURRENT_SOURCE_DIR}/roots.gen.txt)
cw_add_app(beatsync_auto SOURCES ${BS_SRC} PACK ${CMAKE_CURRENT_SOURCE_DIR}/roots.gen.txt)
target_compile_definitions(beatsync_auto PRIVATE GAME_AUTOPILOT=1)
add_dependencies(beatsync_auto_pack beatsync_pack) # the packs share an import cache
